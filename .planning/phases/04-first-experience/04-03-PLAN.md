```ts
/** The pad is nine by nine. The canvas backing store is exactly that. */
export const GRID_SIDE = 9;

/** One ImageData per canvas, created once and written in place on every paint. */
export function createScratch(ctx: CanvasRenderingContext2D): ImageData {
  return ctx.createImageData(GRID_SIDE, GRID_SIDE);
}

/**
 * Paint one pad's frame. `frame` is PadSim.frame: 243 bytes, screen order, RGB.
 * The canvas is 9x9 and CSS scales it with image-rendering: pixelated, so the
 * compositor does the upscale for free and this is the only draw call.
 */
export function paintPad(
  ctx: CanvasRenderingContext2D,
  frame: Uint8Array,
  scratch: ImageData,
): void;
```

The loop, and nothing else:

```ts
const px = scratch.data;
for (let n = 0; n < GRID_SIDE * GRID_SIDE; n++) {
  const r = frame[n * 3], g = frame[n * 3 + 1], b = frame[n * 3 + 2];
  const lit = r !== 0 || g !== 0 || b !== 0;
  px[n * 4] = r; px[n * 4 + 1] = g; px[n * 4 + 2] = b;
  px[n * 4 + 3] = lit ? 255 : 0;
}
ctx.putImageData(scratch, 0, 0);
```

```ts
export const MAX_CONTACTS = 5;

/** Canvas offset to an LED coordinate. See the comment for why the factor is max + 1. */
export function mapAxis(offset: number, extent: number, max: number): number;

/** The minimum an engine must offer for a finger to reach it. PadSim satisfies it. */
export interface TouchTarget {
  touchDown(id: number, x: number, y: number): void;
  touchMove(id: number, x: number, y: number): void;
  touchUp(id: number, x: number, y: number): void;
}

export class TouchSampler {
  /** Number of live contacts. The host uses it to decide the pad is active. */
  get size(): number;
  down(pointerId: number, x: number, y: number): boolean;   // false when no slot is free
  move(pointerId: number, x: number, y: number): void;
  end(pointerId: number): void;
  /** At most one sample per contact. Called once per tick, by the host. */
  deliver(target: TouchTarget): void;
  clear(): void;
}
```

Reproduce the vendored semantics exactly, and carry the comments that explain them:

- MOVE coalescing: if the last pending sample is a MOVE, overwrite its coordinates rather than
  queueing a second one. DOWN and UP always keep their place.

  contact's final sample.
