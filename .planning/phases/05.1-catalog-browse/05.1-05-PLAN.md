same class as `identity.spec.ts`'s X-27 note for Phase 5's ninth token, and each carries the
amendment note in the file's own header rather than only in a SUMMARY.

```typescript
// D-07: one page per CATALOG entry, not one per row entry. The import is still
// the LIGHT module - listing.ts imports nothing at runtime, exactly as
// front-door.ts does - because whatever this file imports lands in the client
// bundle of a page whose whole job is to paint quickly. config-shape.spec.ts
// test 13 (source) and test 14 (built artefact) both hold that line.
import { LISTING, listingIndex } from "$lib/catalog/listing";

export const prerender = true;

/** Sixteen pages. Trackpad included (D-11): its page renders the dark pad honestly. */
export function entries() {
  return LISTING.map((entry) => ({ id: entry.id }));
}

export function load({ params }: { params: { id: string } }) {
  return { id: params.id, index: listingIndex(params.id) };
}
```
