## `src/lib/catalog/calibration.ts` - the contract every later plan imports

```ts
/** Nine knots per axis: the raw 0..127 value the sensor reports with a finger centred on LED n. */
export const KX: readonly number[];   // [1, 13, 29, 46, 64, 85, 100, 120, 126]
export const KY: readonly number[];   // [0, 12, 26, 48, 68, 83, 97, 115, 126]
export const MEASURED = true as const;
export const CALIBRATION_PROVENANCE: {
  readonly date: "2026-09-11";
  readonly source: "BENCH-2026-09-11.txt, Probe C run 1, 11:43:50-11:44:31";
  readonly runs: 1;
  readonly discarded: "target 10 (cell 4): (127, 68) is x pinned at the right edge with y on row 4 - a re-tap after a dropped lift at target 9";
  readonly ky0: "corners 18 (1,0) and 19 (127,0), both y = 0";
  readonly corners: readonly [readonly [number, number], ...];  // the four corner triples as read
  readonly handAndFlatness: "not reported";
};
/** `KX={1,13,...}KY={0,12,...}` - the exact literal library.ts renders into 255/0. No spaces. */
export function renderKnots(): string;
/** The FORWARD map: LED coordinate u (0..8, LED n at u = n, fractional between) -> raw sensor value 0..127
 *  (integer, floored), piecewise-linear over the knots, saturating at the outer knots below 0 and above 8.
 *  `max` scales for hi-res: 127 -> as is; 1023 -> x8 (the firmware's own txma(1023) identity). */
export function sensorAt(u: number, axis: "x" | "y", max?: number): number;
/** The TS twin of the Lua `U(v, k)`: raw 0..127 -> 0..512 (LED n at n*64), the SAME integer arithmetic
 *  as the Lua including floor division, so a test can compute the cell and the weights G will produce. */
export function calibratedAxis(v: number, axis: "x" | "y"): number;
export const LED_STEP = 64;
```

why; (b) that a second pass replaces the two arrays and nothing else, then `UPDATE_FRAMES=1` regenerates
`frames.json` (plan 05's command); (c) that the module imports nothing (like `snapshot.ts`, so
`library.ts`, `touch.ts` and `demo.ts` can all import it without pulling anything onto the first paint);
(d) that the map is of ONE prototype unit and another module may differ - the sentence 13's copy pass
