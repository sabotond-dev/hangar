export interface QueueOptions {
  preSendDelayMs?: number;                   // default PRE_SEND_DELAY_MS (10); the A/B sets it to 0
  attempts?: number;                         // default RETRY_ATTEMPTS (3)
  backoffMs?: (attempt: number) => number;   // default retryBackoffMs

  onStep?: (step: CaptureStep) => void;
}

export class RequestQueue {
  constructor(transport: GridTransport, opts?: QueueOptions);
  /** Enqueued, one outstanding at a time. Resolves on a matching response only. */
  request(req: GridRequest): Promise<DecodedClass>;
  /** Fire and forget, jumps the queue, never waits. The restore heartbeat uses this. */
  sendImmediate(req: GridRequest): Promise<void>;
  /** Called by the frame pump for every decoded class in every frame. */

  abort(reason: string): void;
}

Rules, each carrying its provenance in a comment:

- **One outstanding request**: a single instance field plus a promise chain. Never a busy-wait — the
  desktop's `while (...) await sleep(1)` becomes one second per iteration in a throttled tab
  (PITFALLS C11).
- **Pre-send gap**: `await sleep(preSendDelayMs)` before each write, configurable and settable to 0
  for the A/B.
- **HEARTBEAT never resolves a waiter**, checked first — `matchResponse` already does this, and
  `deliver` must not shortcut around it.
- **NACK rejects immediately and is never retried** — on `matchResponse === "nack"`, reject without
  spending an attempt.
- **Bounded retry**: `attempts` total writes, then a named failure. Never the desktop's unbounded
  recursion.
- **Deadlines from `now()`, not from the timer's duration.** A hidden tab throttles timers to 1 Hz, so
  a `setTimeout(500)` can fire at 1000 ms and report a timeout that never happened. On fire, re-check
  `now() - startedAt`; if it is short, re-arm for the remainder.
